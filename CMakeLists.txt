cmake_minimum_required(VERSION 3.22)
project(SolarSandbox VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(SOLAR_BUILD_EXTENSION "Build Godot GDExtension" ON)
add_library(solar_core STATIC native/core/physics.cpp)
target_include_directories(solar_core PUBLIC native/core)
if(MSVC)
  target_compile_options(solar_core PRIVATE /W4 /fp:precise /utf-8)
else()
  target_compile_options(solar_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(physics_tests native/tests/test_physics.cpp)
  target_link_libraries(physics_tests PRIVATE solar_core)
  add_test(NAME physics_validation COMMAND physics_tests)
endif()
if(SOLAR_BUILD_EXTENSION)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/godot-cpp/CMakeLists.txt")
    message(FATAL_ERROR "Run tools/bootstrap.py first (godot-cpp 4.4.1).")
  endif()
  # The same extension is used by the editor and exported release. Release bindings
  # also work in the editor; debug bindings cannot use release engine memory padding.
  set(GODOTCPP_TARGET template_release CACHE STRING "Godot binding target" FORCE)
  set(GODOTCPP_PRECISION single CACHE STRING "Godot render precision")
  set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/native/build_profile.json" CACHE FILEPATH "")
  set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "")
  add_subdirectory(third_party/godot-cpp)
  add_library(solar_physics SHARED native/extension/solar_simulation.cpp native/extension/register_types.cpp)
  target_link_libraries(solar_physics PRIVATE solar_core godot-cpp)
  target_include_directories(solar_physics PRIVATE native/extension)
  if(MSVC)
    target_compile_options(solar_physics PRIVATE /utf-8 /EHsc)
  elseif(MINGW)
    target_link_options(solar_physics PRIVATE -static-libgcc -static-libstdc++)
  endif()
  set_target_properties(solar_physics PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/godot/bin" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/godot/bin")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(solar_physics PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/godot/bin" RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/godot/bin")
  endforeach()
endif()
